cmake_minimum_required(VERSION 3.20)
project(UDFFirmware C)
if(NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64|AMD64)$")
  message(FATAL_ERROR "AMD64 firmware profile required")
endif()
if(NOT EXISTS "${COMMON_INCLUDE_DIR}/infiltratr/filesystem_module_v2.h")
  message(FATAL_ERROR "Pinned canonical Common V2 headers required")
endif()
add_library(firmware MODULE ../../native/filesystems/udf/udf.c)
set_target_properties(firmware PROPERTIES PREFIX "" OUTPUT_NAME "UDF" SUFFIX ".so" C_STANDARD 11 C_STANDARD_REQUIRED ON C_VISIBILITY_PRESET hidden)
target_include_directories(firmware PRIVATE "${COMMON_INCLUDE_DIR}")
target_compile_definitions(firmware PRIVATE IFS_BOOT_FIRMWARE=1)
target_compile_options(firmware PRIVATE -O2 -Wall -Wextra -Werror -march=x86-64 -mtune=generic -mno-red-zone -mgeneral-regs-only -fcf-protection=none -ffreestanding -fno-builtin -fno-stack-protector -fno-tree-loop-distribute-patterns -U_FORTIFY_SOURCE -D_FORTIFY_SOURCE=0)
target_link_options(firmware PRIVATE -nostdlib -Wl,-Bsymbolic,-z,defs,-z,noexecstack,-z,relro,-z,now,-z,nopack-relative-relocs,-z,max-page-size=4096)
